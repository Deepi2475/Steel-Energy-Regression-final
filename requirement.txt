Project overview and plan
This will be a full, production-style ML project for energy consumption regression using the Kaggle “Steel Plant Energy Consumption Data” dataset. We’ll set up a clean repo with a UV-based Python environment, implement robust data cleaning and time-based splitting, run H2O AutoML to select candidate models, train three manual models with MLflow logging, set up a remote MLflow Tracking Server (AWS EC2 + Neon Postgres + S3), register and compare models, deploy a FastAPI prediction service, and analyze drift using a later time window.

High-level steps:

Data ingestion, schema definition, and cleaning

Time-based split (35% train, 35% validate, 30% test) preserving chronological order

H2O AutoML on training set only; select top 3 algorithms by RMSE/MAE

Manual model training scripts for three algorithms with MLflow logging

Remote MLflow Tracking Server setup with Neon + S3 artifact store

Model registry and comparison

FastAPI deployment with three model endpoints

Model drift analysis with Evidently/NannyML

GitHub submission with screenshots and code

Data ingestion and cleaning
We’ll implement robust cleaning:

Parse combined date/time into a single pandas datetime column

Handle missing values: numeric impute with median; categorical impute with most frequent

Fix outliers: cap numeric columns using IQR-based winsorization or domain-informed thresholds

Normalize text fields: strip, lower-case categorical labels

Encode categoricals using OneHotEncoder (handle unknowns)

Persist a feature schema and preprocessing pipeline artifact

Configuration: config/model_config.yaml

Ingestion and cleaning script: scripts/01_ingest_and_clean.py
python
import os
import re
import pandas as pd
import numpy as np
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
import yaml

DATA_RAW = "data/raw/steel_energy.csv"
DATA_DIR = "data/processed"
PIPELINE_PATH = os.path.join(DATA_DIR, "preprocess_pipeline.pkl")
SCHEMA_PATH = os.path.join(DATA_DIR, "feature_schema.yaml")

def parse_datetime(df, col):
    # The dataset has a combined date/time string like "1/1/2018 0:15"
    df[col] = pd.to_datetime(df[col], errors="coerce", infer_datetime_format=True)
    return df

def normalize_text(df, cols):
    for c in cols:
        df[c] = df[c].astype(str).str.strip().str.lower()
    return df

def cap_outliers_iqr(df, numeric_cols, factor=3.0):
    for c in numeric_cols:
        q1, q3 = df[c].quantile([0.25, 0.75])
        iqr = q3 - q1
        lower = q1 - factor * iqr
        upper = q3 + factor * iqr
        df[c] = df[c].clip(lower, upper)
    return df

def load_config():
    with open("config/model_config.yaml", "r") as f:
        return yaml.safe_load(f)

def main():
    os.makedirs(DATA_DIR, exist_ok=True)
    cfg = load_config()

    df = pd.read_csv(DATA_RAW)
    # Rename columns if needed to match config
    # Ensure target and datetime exist
    assert cfg["target"] in df.columns, f"Missing target {cfg['target']}"
    assert cfg["datetime_col"] in df.columns, f"Missing datetime {cfg['datetime_col']}"

    # Parse datetime
    df = parse_datetime(df, cfg["datetime_col"])
    df = df.sort_values(cfg["datetime_col"]).reset_index(drop=True)

    # Normalize text fields
    df = normalize_text(df, cfg["categorical_cols"])

    # Basic cleaning: remove impossible negatives for certain features (domain-aware)
    non_negative = [cfg["target"], "CO2(tCO2)", "NSM"]
    for c in non_negative:
        if c in df.columns:
            df.loc[df[c] < 0, c] = np.nan

    # Outlier capping
    if cfg.get("outlier_cap", {}).get("method") == "iqr":
        df = cap_outliers_iqr(df, cfg["numeric_cols"], factor=cfg["outlier_cap"].get("factor", 3.0))

    # Build preprocessing pipeline for features (exclude target, datetime)
    feature_cols = cfg["numeric_cols"] + cfg["categorical_cols"]
    X = df[feature_cols]
    y = df[cfg["target"]]

    num_imputer = SimpleImputer(strategy="median")
    cat_imputer = SimpleImputer(strategy="most_frequent")
    encoder = OneHotEncoder(handle_unknown="infrequent_if_exist", min_frequency=0.01, sparse_output=False)

    preprocessor = ColumnTransformer(
        transformers=[
            ("num", num_imputer, cfg["numeric_cols"]),
            ("cat", Pipeline(steps=[("impute", cat_imputer), ("encode", encoder)]), cfg["categorical_cols"]),
        ],
        remainder="drop",
        verbose_feature_names_out=True,
    )

    # Fit the preprocessor on full dataset (fits encoders/imputers, but models will be trained on train split later)
    preprocessor.fit(X)

    # Persist pipeline and schema
    import joblib
    joblib.dump(preprocessor, PIPELINE_PATH)

    schema = {
        "datetime_col": cfg["datetime_col"],
        "target": cfg["target"],
        "numeric_cols": cfg["numeric_cols"],
        "categorical_cols": cfg["categorical_cols"],
        "feature_cols": feature_cols,
    }
    with open(SCHEMA_PATH, "w") as f:
        yaml.safe_dump(schema, f)

    # Save a cleaned full CSV for splitting
    df.to_csv(os.path.join(DATA_DIR, "cleaned_full.csv"), index=False)
    print("Cleaned dataset and preprocessing pipeline saved.")

if __name__ == "__main__":
    main()
Time-based split and saving files
We’ll split by chronological proportion of the timeframe (not row count): 35% train, 35% validation, 30% test.

Split script: scripts/02_time_split.py
python
import os
import pandas as pd
import yaml

DATA_DIR = "data/processed"

def main():
    with open(os.path.join(DATA_DIR, "feature_schema.yaml"), "r") as f:
        schema = yaml.safe_load(f)
    dt_col = schema["datetime_col"]

    df = pd.read_csv(os.path.join(DATA_DIR, "cleaned_full.csv"))
    df[dt_col] = pd.to_datetime(df[dt_col])
    df = df.sort_values(dt_col)

    t_min, t_max = df[dt_col].min(), df[dt_col].max()
    total_range = (t_max - t_min).total_seconds()
    t_train_end = t_min + pd.to_timedelta(total_range * 0.35, unit="s")
    t_val_end = t_min + pd.to_timedelta(total_range * 0.70, unit="s")  # 35% train + 35% val

    train_df = df[df[dt_col] <= t_train_end]
    val_df = df[(df[dt_col] > t_train_end) & (df[dt_col] <= t_val_end)]
    test_df = df[df[dt_col] > t_val_end]

    os.makedirs(DATA_DIR, exist_ok=True)
    train_df.to_csv(os.path.join(DATA_DIR, "train.csv"), index=False)
    val_df.to_csv(os.path.join(DATA_DIR, "validate.csv"), index=False)
    test_df.to_csv(os.path.join(DATA_DIR, "test.csv"), index=False)
    print(f"Saved splits: train={len(train_df)}, val={len(val_df)}, test={len(test_df)}")

if __name__ == "__main__":
    main()
H2O AutoML and top model selection
We’ll run H2O AutoML on the training set only, using features processed by our pipeline and target. We’ll score on validation set and pick top three model types by RMSE/MAE.

AutoML script: scripts/03_run_h2o_automl.py
python
import os
import yaml
import pandas as pd
import joblib
import h2o
from h2o.automl import H2OAutoML
from sklearn.metrics import mean_absolute_error, mean_squared_error

DATA_DIR = "data/processed"
RESULTS_DIR = "reports"

def main():
    os.makedirs(RESULTS_DIR, exist_ok=True)
    with open(os.path.join(DATA_DIR, "feature_schema.yaml"), "r") as f:
        schema = yaml.safe_load(f)

    dt_col = schema["datetime_col"]; target = schema["target"]
    feature_cols = schema["feature_cols"]
    pre = joblib.load(os.path.join(DATA_DIR, "preprocess_pipeline.pkl"))

    # Load sets
    train = pd.read_csv(os.path.join(DATA_DIR, "train.csv"))
    val = pd.read_csv(os.path.join(DATA_DIR, "validate.csv"))

    # Preprocess features
    X_train = train[feature_cols]; y_train = train[target]
    X_val = val[feature_cols]; y_val = val[target]

    X_train_t = pre.transform(X_train)
    X_val_t = pre.transform(X_val)

    # H2O setup
    h2o.init()
    import numpy as np
    # Build H2O frames from transformed numpy; we’ll attach target separately
    x_cols = [f"f_{i}" for i in range(X_train_t.shape[1])]
    train_h2o = h2o.H2OFrame(pd.DataFrame(X_train_t, columns=x_cols))
    train_h2o[target] = h2o.H2OFrame(y_train.to_numpy().reshape(-1, 1))
    val_h2o = h2o.H2OFrame(pd.DataFrame(X_val_t, columns=x_cols))
    val_h2o[target] = h2o.H2OFrame(y_val.to_numpy().reshape(-1, 1))

    aml = H2OAutoML(max_runtime_secs=1800, seed=42, sort_metric="RMSE", exclude_algos=None)
    aml.train(x=x_cols, y=target, training_frame=train_h2o, leaderboard_frame=val_h2o)

    lb = aml.leaderboard.as_data_frame()
    lb.to_csv(os.path.join(RESULTS_DIR, "h2o_leaderboard.csv"), index=False)

    # Pick top 3 distinct algorithm types from model_id strings
    # Example model_ids: "XGBoost_...", "GBM_...", "DRF_...", "StackedEnsemble_..."
    def model_type(mid):
        if mid.startswith("XGBoost"): return "xgboost"
        if mid.startswith("GBM"): return "lightgbm_like"  # H2O GBM
        if mid.startswith("DRF"): return "random_forest"
        if mid.startswith("GLM"): return "glm"
        if mid.startswith("DeepLearning"): return "dnn"
        if mid.startswith("StackedEnsemble"): return "stacked_ensemble"
        return "other"

    selected = []
    seen = set()
    for _, row in lb.iterrows():
        mtype = model_type(row["model_id"])
        if mtype not in seen and mtype not in {"stacked_ensemble", "other"}:
            selected.append({"model_id": row["model_id"], "type": mtype, "rmse": row["rmse"], "mae": row.get("mae", None)})
            seen.add(mtype)
        if len(selected) == 3:
            break

    pd.DataFrame(selected).to_csv(os.path.join(RESULTS_DIR, "selected_models.csv"), index=False)
    print("Selected top 3 algorithm types:", [s["type"] for s in selected])

if __name__ == "__main__":
    main()

python
import os, yaml, joblib, json
import pandas as pd
import numpy as np
from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score
import mlflow
import mlflow.sklearn
from xgboost import XGBRegressor
import matplotlib.pyplot as plt
import seaborn as sns

EXP_NAME = "steel_energy_models"
DATA_DIR = "data/processed"
ART_DIR = "mlflow_artifacts"

def load_data():
    with open(os.path.join(DATA_DIR, "feature_schema.yaml"), "r") as f:
        schema = yaml.safe_load(f)
    pre = joblib.load(os.path.join(DATA_DIR, "preprocess_pipeline.pkl"))
    feature_cols = schema["feature_cols"]; target = schema["target"]

    def prep(split):
        df = pd.read_csv(os.path.join(DATA_DIR, f"{split}.csv"))
        X = pre.transform(df[feature_cols])
        y = df[target].values
        return X, y
    return schema, pre, prep

def eval_and_log(y_true, y_pred, split_label):
    rmse = np.sqrt(mean_squared_error(y_true, y_pred))
    mae = mean_absolute_error(y_true, y_pred)
    r2 = r2_score(y_true, y_pred)
    mlflow.log_metric(f"{split_label}_rmse", rmse)
    mlflow.log_metric(f"{split_label}_mae", mae)
    mlflow.log_metric(f"{split_label}_r2", r2)
    return rmse, mae, r2

def plot_residuals(y_true, y_pred, name):
    plt.figure(figsize=(6,4))
    resid = y_true - y_pred
    sns.histplot(resid, bins=50, kde=True)
    plt.title(f"Residuals - {name}")
    pth = os.path.join(ART_DIR, f"residuals_{name}.png")
    os.makedirs(ART_DIR, exist_ok=True)
    plt.savefig(pth); plt.close()
    mlflow.log_artifact(pth)

def main():
    # Point to remote MLflow if configured via env MLflow Tracking URI
    mlflow.set_experiment(EXP_NAME)

    schema, pre, prep = load_data()
    X_train, y_train = prep("train")
    X_val, y_val = prep("validate")
    X_test, y_test = prep("test")

    params = {
        "n_estimators": 800,
        "max_depth": 6,
        "learning_rate": 0.03,
        "subsample": 0.8,
        "colsample_bytree": 0.8,
        "random_state": 42,
        "n_jobs": -1,
        "tree_method": "hist",
    }

    with mlflow.start_run(run_name="xgboost_regressor"):
        mlflow.log_params(params)
        model = XGBRegressor(**params)
        model.fit(X_train, y_train, eval_set=[(X_val, y_val)], verbose=False)

        val_pred = model.predict(X_val)
        test_pred = model.predict(X_test)

        eval_and_log(y_val, val_pred, "val")
        eval_and_log(y_test, test_pred, "test")

        plot_residuals(y_val, val_pred, "xgb_val")
        plot_residuals(y_test, test_pred, "xgb_test")

        # Log model (with preprocessing path as artifact)
        mlflow.sklearn.log_model(model, "model", registered_model_name="steel_energy_xgboost")
        mlflow.log_artifact(os.path.join(DATA_DIR, "preprocess_pipeline.pkl"))

    print("XGBoost training completed.")

if __name__ == "__main__":
    main()

from pydantic import BaseModel, Field, conlist
from typing import List, Optional

class FeatureRecord(BaseModel):
    # Use normalized field names and types
    Lagging_Current_Reactive_Power_kVarh: float = Field(..., alias="Lagging_Current_Reactive.Power_kVarh")
    Leading_Current_Reactive_Power_kVarh: float = Field(..., alias="Leading_Current_Reactive_Power_kVarh")

    Lagging_Current_Power_Factor: float
    Leading_Current_Power_Factor: float
    NSM: float
    WeekStatus: str
    Day_of_week: str
    Load_Type: str

class PredictRequest(BaseModel):
    records: List[FeatureRecord]

class PredictResponse(BaseModel):
    model_name: str
    model_version: Optional[str]
    timestamp: str
    predictions: List[float]
Loader: api/loader.py
python
import os, joblib, yaml, mlflow
from mlflow import pyfunc

DATA_DIR = "data/processed"

def load_preprocessor():
    return joblib.load(os.path.join(DATA_DIR, "preprocess_pipeline.pkl"))

def load_model_by_name(name: str, version: str | None = None):
    # If using Registry versions:
    if version:
        uri = f"models:/{name}/{version}"
    else:
        uri = f"models:/{name}/Production"  # or "Staging"
    model = pyfunc.load_model(uri)
    return model
FastAPI app: api/main.py
python
from fastapi import FastAPI, HTTPException
from datetime import datetime, timezone
import os, yaml, pandas as pd
from .schemas import PredictRequest, PredictResponse
from .loader import load_preprocessor, load_model_by_name

import mlflow
from dotenv import load_dotenv

load_dotenv("config/mlflow.env")
mlflow.set_tracking_uri(os.getenv("MLFLOW_TRACKING_URI"))

app = FastAPI(title="Steel Energy Prediction API")

preprocessor = load_preprocessor()

# Helper: convert incoming records to DataFrame with expected columns
def records_to_df(records):
    # Ensure aliases are mapped properly
    rows = []
    for r in records:
        item = r.model_dump(by_alias=True)
        rows.append(item)
    df = pd.DataFrame(rows)
    return df

def predict_common(model_name, model_version, req: PredictRequest):
    try:
        df = records_to_df(req.records)
    except Exception as e:
        raise HTTPException(status_code=400, detail=f"Invalid input: {e}")

    # Preprocess: must match training feature schema order
    with open("data/processed/feature_schema.yaml", "r") as f:
        schema = yaml.safe_load(f)
    X = df[schema["feature_cols"]]
    X_t = preprocessor.transform(X)

    model = load_model_by_name(model_name, model_version)
    preds = model.predict(X_t).tolist()

    return PredictResponse(
        model_name=model_name,
        model_version=model_version,
        timestamp=datetime.now(timezone.utc).isoformat(),
        predictions=preds,
    )

@app.post("/predict_model1", response_model=PredictResponse)
def predict_model1(req: PredictRequest):
    return predict_common("steel_energy_xgboost", None, req)

@app.post("/predict_model2", response_model=PredictResponse)
def predict_model2(req: PredictRequest):
    return predict_common("steel_energy_lightgbm", None, req)

@app.post("/predict_model3", response_model=PredictResponse)
def predict_model3(req: PredictRequest):
    return predict_common("steel_energy_randomforest", None, req)

Model drift analysis
We’ll simulate production drift by using a newer timeframe (e.g., the final months of 2018) and run Evidently/NannyML to analyze data and performance drift.

Evidently script: scripts/08_drift_evidently.py
python
import os, yaml, joblib, pandas as pd
from evidently.report import Report
from evidently.metrics import DataDriftPreset, RegressionPerformanceMetrics

DATA_DIR = "data/processed"

def main():
    with open(os.path.join(DATA_DIR, "feature_schema.yaml"), "r") as f:
        schema = yaml.safe_load(f)
    target = schema["target"]; feature_cols = schema["feature_cols"]
    pre = joblib.load(os.path.join(DATA_DIR, "preprocess_pipeline.pkl"))

    ref = pd.read_csv(os.path.join(DATA_DIR, "validate.csv"))  # reference window
    prod = pd.read_csv(os.path.join(DATA_DIR, "test.csv"))     # pretend production window

    # Preprocess and attach predictions from champion model (assume LightGBM, adjust after results)
    from mlflow.pyfunc import load_model
    model = load_model("models:/steel_energy_lightgbm/Production")

    ref_X = pre.transform(ref[feature_cols])
    prod_X = pre.transform(prod[feature_cols])

    ref["prediction"] = model.predict(ref_X)
    prod["prediction"] = model.predict(prod_X)

    report = Report(metrics=[
        DataDriftPreset(),
        RegressionPerformanceMetrics(target_column=target, prediction_column="prediction")
    ])
    report.run(reference_data=ref[feature_cols + [target, "prediction"]],
               current_data=prod[feature_cols + [target, "prediction"]])

    os.makedirs("reports", exist_ok=True)
    report.save_html("reports/drift_report.html")
    print("Evidently drift report saved to reports/drift_report.html")

if __name__ == "__main__":
    main()
NannyML script: scripts/09_drift_nannyml.py
python
import os, yaml, joblib, pandas as pd
import nannyml as nml

DATA_DIR = "data/processed"

def main():
    with open(os.path.join(DATA_DIR, "feature_schema.yaml"), "r") as f:
        schema = yaml.safe_load(f)
    target = schema["target"]; feature_cols = schema["feature_cols"]
    pre = joblib.load(os.path.join(DATA_DIR, "preprocess_pipeline.pkl"))

    ref = pd.read_csv(os.path.join(DATA_DIR, "validate.csv"))
    prod = pd.read_csv(os.path.join(DATA_DIR, "test.csv"))

    from mlflow.pyfunc import load_model
    model = load_model("models:/steel_energy_lightgbm/Production")

    ref_X = pre.transform(ref[feature_cols]); prod_X = pre.transform(prod[feature_cols])
    ref["y_pred"] = model.predict(ref_X); prod["y_pred"] = model.predict(prod_X)

    # Data drift (Univariate)
    univar = nml.UnivariateDriftCalculator(
        column_names=feature_cols, chunk_size=500, continuous_methods=["ks"], categorical_methods=["chi2"]
    )
    univar.fit(ref[feature_cols])
    univar_res = univar.calculate(prod[feature_cols])
    univar_res.to_df().to_csv("reports/nannyml_univariate_drift.csv", index=False)

    # Performance drift (with ground truth available)
    pd_calc = nml.PerformanceCalculator(
        y_pred_proba=None, y_pred="y_pred", y_true=target, metrics=["mae", "rmse", "r2"], chunk_size=500
    )
    pd_calc.fit(ref[[target, "y_pred"]])
    pd_res = pd_calc.calculate(prod[[target, "y_pred"]])
    pd_res.to_df().to_csv("reports/nannyml_performance_drift.csv", index=False)

    print("NannyML drift outputs saved to reports/")

if __name__ == "__main__":
    main()
